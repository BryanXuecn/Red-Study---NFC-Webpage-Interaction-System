### 1. 时空对比滑动实现
```javascript
class ImageComparison {
    constructor(container) {
        this.container = container;
        this.slider = container.querySelector('#comparison-range');
        this.beforeImage = container.querySelector('.before-image');
        this.afterImage = container.querySelector('.after-image');
        
        this.init();
    }
    
    init() {
        this.slider.addEventListener('input', (e) => {
            const value = e.target.value;
            this.updateComparison(value);
        });
        
        // 支持触摸滑动
        this.addTouchSupport();
    }
    
    updateComparison(percentage) {
        this.beforeImage.style.clipPath = `polygon(0 0, ${percentage}% 0, ${percentage}% 100%, 0 100%)`;
        
        const sliderHandle = this.container.querySelector('.slider-handle');
        sliderHandle.style.left = `${percentage}%`;
    }
    
    addTouchSupport() {
        // 触摸手势支持代码
        let isDragging = false;
        
        this.container.addEventListener('touchstart', (e) => {
            isDragging = true;
            this.updateFromTouch(e);
        });
        
        this.container.addEventListener('touchmove', (e) => {
            if (isDragging) {
                this.updateFromTouch(e);
            }
        });
        
        this.container.addEventListener('touchend', () => {
            isDragging = false;
        });
    }
}
```

### 2. 3D城市模型实现
```javascript
class CityModel3D {
    constructor(canvasElement) {
        this.canvas = canvasElement;
        this.scene = new THREE.Scene();
        this.camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000);
        this.renderer = new THREE.WebGLRenderer({ canvas: this.canvas, antialias: true });
        
        this.buildings = [];
        this.raycaster = new THREE.Raycaster();
        this.mouse = new THREE.Vector2();
        
        this.init();
    }
    
    init() {
        this.setupRenderer();
        this.setupLighting();
        this.createBuildings();
        this.setupControls();
        this.addEventListeners();
        this.animate();
    }
    
    setupRenderer() {
        this.renderer.setSize(this.canvas.offsetWidth, this.canvas.offsetHeight);
        this.renderer.setClearColor(0x87CEEB, 1); // 天空蓝
        this.renderer.shadowMap.enabled = true;
        this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    }
    
    createBuildings() {
        const buildingData = [
            { x: -5, y: 0, z: -5, width: 2, height: 8, depth: 2, color: 0x8b5cf6, name: '创新大厦' },
            { x: 0, y: 0, z: -3, width: 3, height: 12, depth: 3, color: 0x22c55e, name: '生态中心' },
            { x: 4, y: 0, z: -2, width: 2.5, height: 10, depth: 2.5, color: 0x0ea5e9, name: '智慧大厦' },
            { x: -2, y: 0, z: 3, width: 1.5, height: 6, depth: 1.5, color: 0xf59e0b, name: '文化馆' },
            { x: 3, y: 0, z: 4, width: 2, height: 9, depth: 2, color: 0xdc2626, name: '艺术中心' }
        ];
        
        buildingData.forEach(data => {
            const geometry = new THREE.BoxGeometry(data.width, data.height, data.depth);
            const material = new THREE.MeshLambertMaterial({ color: data.color });
            const building = new THREE.Mesh(geometry, material);
            
            building.position.set(data.x, data.height / 2, data.z);
            building.castShadow = true;
            building.receiveShadow = true;
            building.userData = { name: data.name, height: data.height, function: '多功能建筑' };
            
            this.scene.add(building);
            this.buildings.push(building);
        });
    }
    
    addEventListeners() {
        // 鼠标点击选择建筑
        this.canvas.addEventListener('click', (event) => {
            this.handleBuildingClick(event);
        });
        
        // 窗口大小变化
        window.addEventListener('resize', () => {
            this.handleResize();
        });
    }
    
    handleBuildingClick(event) {
        const rect = this.canvas.getBoundingClientRect();
        this.mouse.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
        this.mouse.y = -((event.clientY - rect.top) / rect.height) * 2 + 1;
        
        this.raycaster.setFromCamera(this.mouse, this.camera);
        const intersects = this.raycaster.intersectObjects(this.buildings);
        
        if (intersects.length > 0) {
            const building = intersects[0].object;
            this.showBuildingInfo(building.userData);
        }
    }
    
    showBuildingInfo(buildingData) {
        const infoPanel = document.getElementById('building-info');
        document.getElementById('building-name').textContent = buildingData.name;
        document.getElementById('building-height').textContent = buildingData.height + 'm';
        document.getElementById('building-function').textContent = buildingData.function;
        
        infoPanel.classList.remove('hidden');
    }
}
```

### 3. 时间轴动画系统
```javascript
class TimelineAnimation {
    constructor(container) {
        this.container = container;
        this.items = container.querySelectorAll('.timeline-item');
        this.currentIndex = 0;
        this.isPlaying = false;
        this.playInterval = null;
        
        this.init();
    }
    
    init() {
        this.bindControls();
        this.setupIntersectionObserver();
    }
    
    bindControls() {
        const playBtn = document.getElementById('play-timeline');
        const resetBtn = document.getElementById('reset-timeline');
        
        playBtn.addEventListener('click', () => {
            if (this.isPlaying) {
                this.pauseTimeline();
            } else {
                this.playTimeline();
            }
        });
        
        resetBtn.addEventListener('click', () => {
            this.resetTimeline();
        });
    }
    
    playTimeline() {
        this.isPlaying = true;
        this.updatePlayButton();
        
        this.playInterval = setInterval(() => {
            this.activateTimelineItem(this.currentIndex);
            this.currentIndex++;
            
            if (this.currentIndex >= this.items.length) {
                this.pauseTimeline();
                this.currentIndex = 0;
            }
        }, 2000);
    }
    
    activateTimelineItem(index) {
        // 移除所有活动状态
        this.items.forEach(item => item.classList.remove('active'));
        
        // 激活当前项
        if (this.items[index]) {
            this.items[index].classList.add('active');
```

```javascript
    let startX, currentX;

        startX = e.touches[0].clientX;
    });
    
    this.container.addEventListener('touchmove', (e) => {
        currentX = e.touches[0].clientX;
        const rect = this.container.getBoundingClientRect();
        const percentage = ((currentX - rect.left) / rect.width) * 100;
        this.updateComparison(Math.max(0, Math.min(100, percentage)));
    });
}
```
