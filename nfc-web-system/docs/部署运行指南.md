## 🚀 快速开始

### 环境要求
- **操作系统**: Windows 10/11, macOS, Linux
- **Python**: 3.6+ (用于本地开发服务器)
- **浏览器**: Chrome 89+, Safari 14+, Firefox 90+
- **网络**: WiFi网络 (手机和电脑同一网络)

### 本地开发部署

#### 1. 启动开发服务器

### NFC交互方式说明

#### 使用流程
1. **手机靠近NFC芯片**: 用户使用支持NFC的手机靠近物理的NFC芯片/标签
2. **自动识别跳转**: 手机NFC感应到芯片后，自动跳转到对应的网页
3. **网页交互体验**: 网页加载完成后，用户可以进行各种交互操作

#### 支持的NFC标签格式
```javascript
// 推荐使用URL格式（NDEF记录）
```

## 🌐 生产环境部署

### HTTPS部署（NFC功能必需）

#### 使用Nginx + Let's Encrypt
```nginx
# /etc/nginx/sites-available/red-study
server {
    listen 443 ssl http2;
    server_name your-domain.com;
    
    # SSL证书配置
    ssl_certificate /etc/letsencrypt/live/your-domain.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/your-domain.com/privkey.pem;
    
    # 安全配置
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-RSA-AES256-GCM-SHA512:DHE-RSA-AES256-GCM-SHA512;
    ssl_prefer_server_ciphers off;
    
    # 项目文件路径
    root /var/www/red-study/NFC网页交互系统;
    index index.html;
    
    # 静态资源缓存
    location ~* \.(css|js|png|jpg|jpeg|gif|svg|woff2)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        add_header Vary Accept-Encoding;
    }
    
    # 音频文件配置
    location ~* \.(mp3|ogg|wav)$ {
        expires 7d;
        add_header Cache-Control "public";
        add_header Accept-Ranges bytes;
    }
    
    # HTML文件不缓存
    location ~* \.html$ {
        expires 0;
        add_header Cache-Control "no-cache, no-store, must-revalidate";
        add_header Pragma no-cache;
    }
    
    # Gzip压缩
    gzip on;
    gzip_types
        text/plain
        text/css
        text/js
        text/xml
        text/javascript
        application/javascript
        application/xml+rss
        application/json;
}

# HTTP重定向到HTTPS
server {
    listen 80;
    server_name your-domain.com;
    return 301 https://$server_name$request_uri;
}
```

#### 部署脚本
```bash
#!/bin/bash
# deploy.sh

# 项目部署脚本
PROJECT_DIR="/var/www/red-study"
BACKUP_DIR="/backup/red-study-$(date +%Y%m%d%H%M%S)"

echo "开始部署红色研学NFC交互系统..."

# 1. 备份现有版本
if [ -d "$PROJECT_DIR" ]; then
    echo "备份现有版本到 $BACKUP_DIR"
    cp -r "$PROJECT_DIR" "$BACKUP_DIR"
fi

# 2. 同步新代码
echo "同步新代码..."
rsync -av --exclude='.git' --exclude='node_modules' ./NFC网页交互系统/ "$PROJECT_DIR/"

# 3. 设置权限
echo "设置文件权限..."
chown -R www-data:www-data "$PROJECT_DIR"
chmod -R 755 "$PROJECT_DIR"

# 4. 重启Nginx
echo "重启Nginx..."
systemctl reload nginx

# 5. 验证部署
echo "验证部署状态..."
curl -f https://your-domain.com/ > /dev/null

if [ $? -eq 0 ]; then
    echo "✅ 部署成功！"
    echo "访问地址: https://your-domain.com"
else
    echo "❌ 部署失败，恢复备份..."
    cp -r "$BACKUP_DIR" "$PROJECT_DIR"
    systemctl reload nginx
fi
```

### CDN配置

#### 静态资源CDN
```javascript
// config.js 中的CDN配置
window.AppConfig = {
  cdn: {
    enabled: true,
    baseUrl: 'https://cdn.your-domain.com',
    assets: {
      images: '/images',
      audio: '/audio',
      fonts: '/fonts'
    }
  }
};

// 资源URL生成函数
function getAssetUrl(type, filename) {
  const config = window.AppConfig;
  if (config.cdn.enabled) {
    return `${config.cdn.baseUrl}${config.cdn.assets[type]}/${filename}`;
  }
  return `./assets/${type}/${filename}`;
}
```

### 性能监控

#### 关键指标监控
```javascript
// 性能监控代码（插入到所有页面）
(function() {
  // 页面加载时间
  window.addEventListener('load', function() {
    const loadTime = performance.timing.loadEventEnd - performance.timing.navigationStart;
    
    // 发送监控数据
    if (loadTime > 3000) {
      console.warn(`页面加载时间过长: ${loadTime}ms`);
    }
    
    // 可选：发送到监控服务
    sendMetric('page_load_time', loadTime);
  });
  
  // NFC使用统计
  if (window.NFCHandler) {
    const originalHandleRead = window.NFCHandler.prototype.handleNFCRead;
    window.NFCHandler.prototype.handleNFCRead = function(event) {
      sendMetric('nfc_scan_success', 1);
      return originalHandleRead.call(this, event);
    };
  }
  
  function sendMetric(name, value) {
    // 这里可以集成Google Analytics, 百度统计等
    if (typeof gtag !== 'undefined') {
      gtag('event', name, { value: value });
    }
  }
})();
```

#### 错误监控
```javascript
// 全局错误捕获
window.addEventListener('error', function(event) {
  const errorInfo = {
    message: event.message,
    filename: event.filename,
    lineno: event.lineno,
    colno: event.colno,
    stack: event.error?.stack,
    userAgent: navigator.userAgent,
    timestamp: new Date().toISOString()
  };
  
  // 发送错误报告
  sendErrorReport(errorInfo);
});

// Promise错误捕获
window.addEventListener('unhandledrejection', function(event) {
  const errorInfo = {
    message: event.reason.message || event.reason,
    stack: event.reason.stack,
    type: 'promise_rejection',
    timestamp: new Date().toISOString()
  };
  
  sendErrorReport(errorInfo);
});

function sendErrorReport(errorInfo) {
  // 可以发送到Sentry, Bugsnag等错误监控服务
  fetch('/api/errors', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(errorInfo)
  }).catch(console.error);
}
```

## 🔧 故障排除

### 常见问题解决

#### 1. 服务器启动失败
```bash
# 检查端口占用
netstat -an | grep :8080

# 杀死占用进程
lsof -ti:8080 | xargs kill -9

# 重新启动
python -m http.server 8080
```

#### 2. 手机无法访问
```bash
# 检查防火墙设置（Windows）
netsh advfirewall firewall add rule name="Python HTTP Server" dir=in action=allow protocol=TCP localport=8080

# 检查网络连接
ping 192.168.1.109
```

#### 3. NFC跳转功能不工作
- ✅ 确认NFC芯片已正确写入URL
- ✅ 检查手机NFC功能已开启
- ✅ 确认网页URL可正常访问（建议使用HTTPS）
- ✅ 测试芯片距离（建议1-3cm内）
- ✅ 检查芯片格式是否为NDEF

#### 4. 音频播放问题
```javascript
// 检查音频格式支持
```
